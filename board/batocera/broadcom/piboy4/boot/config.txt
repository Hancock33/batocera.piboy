[rpi4]

#Base Settings

initial_turbo=60                  # turbo clocks from boot for the first 60 seconds

disable_splash=1                  # rainbow splash screen will not be shown on boot

#Enable DPI gpio

#Audio Settings

# HDMI Settings

hdmi_mode:1=82                    # DMT 82=1080p@60hz, 85=720p@60hz
                                  # CEA 97=2160p@60Hz, 16=1080p@60Hz, 4=720p@60Hz
hdmi_drive:1=2                    # 1=Normal DVI mode (no sound), 2=Normal HDMI mode and sound
hdmi_force_hotplug:1=1            # pretend an HDMI display is attached
#hdmi_safe=1                      # uncomment if you get no picture on HDMI
#hdmi_enable_4kp60=1              # only enable if you actually output 4K60; it raises clocks/heat
